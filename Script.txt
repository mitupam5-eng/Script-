function UI:MakeDraggable()
    local dragging = false
    local dragStart = Vector2.new()
    local startPos = UDim2.new()
    self.TitleBar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = self.Window.Position
            UserInputService.MouseIconEnabled = false
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            self.Window.Position = UDim2.new(
                startPos.X.Scale,
                startPos.X.Offset + delta.X,
                startPos.Y.Scale,
                startPos.Y.Offset + delta.Y
            )
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
            UserInputService.MouseIconEnabled = true
        end
    end)
end

function UI:ToggleCollapse()
    self.IsCollapsed = not self.IsCollapsed
    if self.IsCollapsed then
        for _, section in ipairs(self.Sections) do
            if section.IsExpanded then
                section.IsExpanded = false
                section.ToggleBtn.Text = "+"
                TweenService:Create(section.Content, TweenInfo.new(0.2), {
                    Size = UDim2.new(1, 0, 0, 0)
                }):Play()
                self.CurrentY = self.CurrentY - section.ContentY
            end
        end
        TweenService:Create(self.Window, TweenInfo.new(0.2), {
            Size = UDim2.new(0, 160, 0, 35)
        }):Play()
        self.CollapseBtn.Text = "+"
        self.ContentContainer.Visible = false
    else
        self.CurrentY = 0
        local currentOffset = 0
        for _, section in ipairs(self.Sections) do
            section.Frame.Position = UDim2.new(0, 0, 0, currentOffset)
            section.Content.Position = UDim2.new(0, 0, 0, currentOffset + 35)
            currentOffset = currentOffset + 35
            self.CurrentY = self.CurrentY + 35
        end
        self.MenuContainer.Size = UDim2.new(1, 0, 0, self.CurrentY)
        local newHeight = math.max(160, 30 + self.CurrentY + 5)
        TweenService:Create(self.Window, TweenInfo.new(0.2), {
            Size = UDim2.new(0, 160, 0, newHeight)
        }):Play()
        self.CollapseBtn.Text = "-"
        task.wait(0.2)
        self.ContentContainer.Visible = true
    end
end

        local buttonIndex = #section.Buttons
        local row = math.floor(buttonIndex / 2)
        local col = buttonIndex % 2

        btn.BackgroundColor3 = Color3.fromRGB(44, 44, 50)

    end)

    return section

local function autoChangeMap()
    pcall(function()
        -- Kích hoạt sự kiện đổi Server / Map từ RemoteEvent
        if Events:FindFirstChild("SetServerSettings") then
            Events.SetServerSettings:FireServer({{ MapTime = -1, Paused = true }})
        end
        
        -- Duyệt tìm UI Nút bấm đổi Map (Skip / Teleport / Next Map)
        local mainGui = PlayerGui:FindFirstChild("MainGui") or PlayerGui:FindFirstChild("GameGui")
        if mainGui then
            for _, v in ipairs(mainGui:GetDescendants()) do
                if v:IsA("TextButton") or v:IsA("ImageButton") then
                    local name = string.lower(v.Name)
                    local text = v:IsA("TextButton") and string.lower(v.Text) or ""
                    if string.find(name, "skip") or string.find(name, "next") or string.find(name, "map") or 
                       string.find(text, "skip") or string.find(text, "next") or string.find(text, "đổi map") then
                        for _, conn in pairs(getconnections(v.MouseButton1Click)) do
                            conn:Fire()
                        end
                    end
                end
            end
        end
    end)
end

local autofarm = false
local autoClaimRewards = false
local autoeat = false

autoSection:CreateToggle("AFK", function(enabled)
    autofarm = enabled
    if not autofarm then return end

    task.spawn(function()
        local startTime = tick()
        local lastEatTime = tick()
        local eatTime = 0
        local sellCount = 0
        local timer = 0
        local grabTimer = 0
        local numChunks = 0
        local sellDebounce = false

        local text = Drawing.new("Text")
        text.Outline = true
        text.OutlineColor = Color3.new(0, 0, 0)
        text.Color = Color3.new(1, 1, 1)
        text.Center = false
        text.Position = Vector2.new(64, 64)
        text.Visible = true

        local map = Workspace:FindFirstChild("Map")
        local chunks = Workspace:FindFirstChild("Chunks")
        local bedrock = Workspace:FindFirstChild("Bedrock") or Instance.new("Part")

        local hum, root, size, events, eat, grab, sell, sendTrack, chunk, autoConn

        local function onCharAdd(char)
            numChunks = 0
            hum = char:WaitForChild("Humanoid", 5)
            root = char:WaitForChild("HumanoidRootPart", 5)
            size = char:WaitForChild("Size", 5)
            events = char:WaitForChild("Events", 5)
            
            if events then
                eat = events:WaitForChild("Eat", 5)
                grab = events:WaitForChild("Grab", 5)
                sell = events:WaitForChild("Sell", 5)
            end
            chunk = char:WaitForChild("CurrentChunk", 5)
            sendTrack = char:WaitForChild("SendTrack", 5)

            if not (hum and root) then return end

            autoConn = RunService.Heartbeat:Connect(function(dt)
                if not autofarm then
                    if autoConn then autoConn:Disconnect() end
                    return
                end

                local ran = tick() - startTime
                local hours = math.floor(ran / 3600)
                local minutes = math.floor(ran / 60)
                local seconds = math.floor(ran)

                local eatMinutes = math.floor(eatTime / 60)
                local eatSeconds = math.floor(eatTime)

                local y = bedrock.Position.Y + (bedrock.Size.Y / 2) + hum.HipHeight + (root.Size.Y / 2)

                local maxSize = (LocalPlayer:FindFirstChild("Upgrades") and LocalPlayer.Upgrades:FindFirstChild("MaxSize") and LocalPlayer.Upgrades.MaxSize.Value) or 100
                local sizeAdd = (LocalPlayer:FindFirstChild("Upgrades") and LocalPlayer.Upgrades:FindFirstChild("Multiplier") and LocalPlayer.Upgrades.Multiplier.Value / 100) or 1
                local addAmount = maxSize / math.max(sizeAdd, 0.001)
                local sellTime = addAmount / 2

                local sellMinutes = math.floor(sellTime / 60)
                local sellSeconds = math.floor(sellTime)

                local secondEarn = math.floor(((maxSize + 0.5) ^ 2 - 0.25) / 2 * 100 / math.max(sellTime, 0.001))
                local dayEarn = secondEarn * 60 * 60 * 24

                text.Text = "Run: " .. string.format("%ih%im%is", hours, minutes % 60, seconds % 60)
                    .. "\nEat: " .. string.format("%im%is", eatMinutes % 60, eatSeconds % 60)
                    .. "\nSell: " .. string.format("%im%is", sellMinutes % 60, sellSeconds % 60)
                    .. "\nDaily: " .. tostring(dayEarn)
                    .. "\nChunks: " .. tostring(numChunks)

                hum:ChangeState(Enum.HumanoidStateType.Physics)
                if grab then grab:FireServer() end
                root.Anchored = false
                if eat then eat:FireServer() end
                if sendTrack then sendTrack:FireServer() end

                if chunk and chunk.Value then
                    if timer > 0 then numChunks += 1 end
                    timer = 0
                    grabTimer += dt
                else
                    timer += dt
                    grabTimer = 0
                end

                if timer > 60 then
                    hum.Health = 0
                    timer = 0
                    numChunks = 0
                    autoChangeMap() -- Gọi hàm đổi Map khi thời gian kẹt quá lâu
                end

                if grabTimer > 15 and size then
                    size.Value = maxSize
                end

                if size and (size.Value >= maxSize or timer > 8) then
                    if timer < 8 then
                        if sell then sell:FireServer() end
                        if not sellDebounce then
                            autoChangeMap() -- Thực hiện Đổi Map tự động
                        end
                        sellDebounce = true
                    else
                        autoChangeMap() -- Thực hiện Đổi Map tự động
                    end
                    numChunks = 0
                elseif size and size.Value == 0 then
                    if sellDebounce then
                        local currentEatTime = tick()
                        eatTime = currentEatTime - lastEatTime
                        lastEatTime = currentEatTime
                        sellCount += 1
                    end
                    sellDebounce = false
                end

                root.CFrame = CFrame.new(0, y, 0)
            end)

            hum.Died:Connect(function()
                if autoConn then autoConn:Disconnect() end
                autoChangeMap()
            end)

            if char:FindFirstChild("LocalChunkManager") then char.LocalChunkManager.Enabled = false end
            if char:FindFirstChild("Animate") then char.Animate.Enabled = false end
        end

        if LocalPlayer.Character then
            task.spawn(onCharAdd, LocalPlayer.Character)
        end
        local charAddConn = LocalPlayer.CharacterAdded:Connect(onCharAdd)

        while autofarm do
            task.wait()
            if Workspace:FindFirstChild("Loading") then
                Workspace.Loading:Destroy()
            end
            if map and chunks then
                map.Parent, chunks.Parent = nil, nil
            end
        end

        charAddConn:Disconnect()
        if autoConn then autoConn:Disconnect() end
        if map and chunks then
            map.Parent, chunks.Parent = Workspace, Workspace
        end

        if LocalPlayer.Character then
            local char = LocalPlayer.Character
            if char:FindFirstChildOfClass("Humanoid") then
                char:FindFirstChildOfClass("Humanoid"):ChangeState(Enum.HumanoidStateType.GettingUp)
            end
            if char:FindFirstChild("LocalChunkManager") then char.LocalChunkManager.Enabled = true end
            if char:FindFirstChild("Animate") then char.Animate.Enabled = true end
        end

        text:Remove()
    end)
end)

    task.spawn(function()
        while autoClaimRewards do

            if LocalPlayer:FindFirstChild("TimedRewards") then
                for _, reward in ipairs(LocalPlayer.TimedRewards:GetChildren()) do
                    if reward.Value > 0 then
                        Events.RewardEvent:FireServer(reward)
                    end
                end
            end
            if Events:FindFirstChild("SpinEvent") then
                Events.SpinEvent:FireServer()
            end
        end
    end)
end)

local isMounted = false
local savedSpeed = 16
local savedJump = 50

figureSection:CreateToggle("ghim cố định", function(enabled)
    isMounted = enabled
    local char = LocalPlayer.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    local root = char:FindFirstChild("HumanoidRootPart")
    
    if isMounted and hum and root then
        savedSpeed = hum.WalkSpeed
        savedJump = hum.JumpPower
        hum.WalkSpeed = 0
        hum.JumpPower = 0
        hum.AutoRotate = false
        root.Anchored = true
    elseif hum and root then
        root.Anchored = false
        hum.WalkSpeed = savedSpeed
        hum.JumpPower = savedJump
        hum.AutoRotate = true
    end
end)

    end
end)

            end

local function playEmote(emoteId)
    local char = LocalPlayer.Character
    if char and char:FindFirstChildOfClass("Humanoid") then
        local hum = char:FindFirstChildOfClass("Humanoid")
        local animation = Instance.new("Animation")
        animation.AnimationId = emoteId
        local track = hum:LoadAnimation(animation)
        track:Play()
    end
end
