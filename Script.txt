-- çªå£ææ½
function UI:MakeDraggable()
    local dragging = false
    local dragStart = Vector2.new()
    local startPos = UDim2.new()
    self.TitleBar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = self.Window.Position
            UserInputService.MouseIconEnabled = false
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            self.Window.Position = UDim2.new(
                startPos.X.Scale,
                startPos.X.Offset + delta.X,
                startPos.Y.Scale,
                startPos.Y.Offset + delta.Y
            )
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
            UserInputService.MouseIconEnabled = true
        end
    end)
end

local function formatTime(seconds)
    local hours = math.floor(seconds / 3600)
    local minutes = math.floor((seconds % 3600) / 60)
    local secs = math.floor(seconds % 60)
    return string.format("%02ih%02im%02is", hours, minutes, secs)
end

local function startAutoFarm()
    -- ç¡®ä¿åªæä¸ä¸ªèªå¨å·åç¨å¨è¿è¡
    if autoFarmThread then
        return
    end
    autoFarmThread = coroutine.wrap(function()
        -- åå»ºç»å¾ææ¬
        local text = Drawing.new("Text")
        text.Outline = true
        text.OutlineColor = Color3.new(0, 0, 0)

        text.Visible = true

        local lastEatTime = tick()

        local grabTimer = 0
        local sellDebounce = false
        local sellCount = 0

        local bedrock = Instance.new("Part")

        bedrock.Size = Vector3.new(2048, 1, 2048)

        local map, chunks = workspace:FindFirstChild("Map"), workspace:FindFirstChild("Chunks")
        if map and chunks then
            map.Parent, chunks.Parent = nil, nil
        end
        local numChunks = 0
        -- è§è²ç»ä»¶åé
        local hum, root, size, events, eat, grab, sell, sendTrack, chunk, radius
        -- è§è²å è½½å¤çå½æ°
        local function onCharAdd(char)
            numChunks = 0
            -- ç­å¾è§è²ç»ä»¶å è½½
            hum = char:WaitForChild("Humanoid")
            root = char:WaitForChild("HumanoidRootPart")

            autoConn = RunService.Heartbeat:Connect(function(dt)
                -- å¦æèªå¨å·å³é­ï¼æ­å¼è¿æ¥
                if not autofarm then
                    autoConn:Disconnect()
                    autoConn = nil
                    return
                end
                -- è®¡ç®è§è²Yåæ 
                local y = bedrock.Position.Y + bedrock.Size.Y / 2 + hum.HipHeight + root.Size.Y / 2
                -- åºåæ£æµé»è¾
                if chunk.Value then
                    if timer > 0 then
                        numChunks += 1
                    end
                    timer = 0
                    grabTimer += dt
                else
                    timer += dt
                    grabTimer = 0
                end
                -- æ¯10å¸§æ´æ°ä¸æ¬¡ç»è®¡ä¿¡æ¯
                if math.fmod(tick() * 60, 10) < 1 then
                    local ran = tick() - startTime
                    local hours = math.floor(ran / 60 / 60)
                    local minutes = math.floor(ran / 60)
                    local seconds = math.floor(ran)
                    local eatMinutes = math.floor(eatTime / 60)
                    local eatSeconds = math.floor(eatTime)
                    local sizeAdd = LocalPlayer.Upgrades.Multiplier.Value / 100
                    local addAmount = LocalPlayer.Upgrades.MaxSize.Value / sizeAdd
                    local sellTime = addAmount / 2
                    local sellMinutes = math.floor(sellTime / 60)
                    local sellSeconds = math.floor(sellTime)
                    local secondEarn = math.floor(sizeGrowth(LocalPlayer.Upgrades.MaxSize.Value) / sellTime)
                    local minuteEarn = secondEarn * 60
                    local hourEarn = minuteEarn * 60
                    local dayEarn = hourEarn * 24
                    text.Text = ""
                        .. "\nRuntime: " .. string.format("%ih%im%is", hours, minutes % 60, seconds % 60)
                        .. "\nActive Time: " .. string.format("%im%is", eatMinutes % 60, eatSeconds % 60)
                        .. "\nApprox Time: " .. string.format("%im%is", sellMinutes % 60, sellSeconds % 60)
                        .. "\nPer Day: " .. dayEarn
                        .. "\nChunks: " .. numChunks
                end
                -- æ¯2å¸§åéä¸æ¬¡äºä»¶
                if math.fmod(tick() * 30, 2) < 1 then
                    hum:ChangeState(Enum.HumanoidStateType.Physics)
                    grab:FireServer()
                    root.Anchored = false
                    eat:FireServer()
                    sendTrack:FireServer()
                end
                -- å¦ææåè®¡æ¶å¨è¶è¿15ç§ï¼å¼ºå¶è®¾ç½®å¤§å°ä¸ºæå¤§å¼
                if grabTimer > 15 then
                    size.Value = LocalPlayer.Upgrades.MaxSize.Value
                end
                -- æ£æ¥æ¯å¦éè¦åºå®æåæ¢å°å¾
                if (size.Value >= LocalPlayer.Upgrades.MaxSize.Value) or timer > 8 then
                    if timer < 8 then
                        sell:FireServer()
                        if not sellDebounce then
                            changeMap()
                        end
                        sellDebounce = true
                    else
                        changeMap()
                    end
                    numChunks = 0
                elseif size.Value == 0 then
                    if sellDebounce then
                        local currentEatTime = tick()
                        eatTime = currentEatTime - lastEatTime
                        lastEatTime = currentEatTime
                        sellCount += 1
                    end
                    sellDebounce = false
                end
                -- åºå®è§è²ä½ç½®
                root.CFrame = CFrame.new(0, y, 350)
            end)
            -- è§è²æ­»äº¡å¤ç
            hum.Died:Connect(function()
                if autoConn then
                    autoConn:Disconnect()
                    autoConn = nil
                end
                changeMap()
            end)
            -- ç¦ç¨è§è²å¨ç»
            char:WaitForChild("Animate").Enabled = false
        end
        -- åå§åè§è²å¤ç
        if LocalPlayer.Character then
            task.spawn(onCharAdd, LocalPlayer.Character)
        else
            task.spawn(onCharAdd, LocalPlayer.CharacterAdded:Wait())
        end
        -- çå¬è§è²å è½½äºä»¶
        charAddConn = LocalPlayer.CharacterAdded:Connect(onCharAdd)
        -- ä¸»å¾ªç¯
        while autofarm do
            local dt = task.wait()
            -- éæ¯å è½½çé¢
            if workspace:FindFirstChild("Loading") then
                workspace.Loading:Destroy()
            end
        end
        -- æ¸çå·¥ä½
        if charAddConn then
            charAddConn:Disconnect()
            charAddConn = nil
        end
        if autoConn then
            autoConn:Disconnect()
            autoConn = nil
        end
        if map and chunks then
            map.Parent, chunks.Parent = workspace, workspace
        end
        if hum then
            hum:ChangeState(Enum.HumanoidStateType.GettingUp)
            task.wait(0.1)
            hum:ChangeState(Enum.HumanoidStateType.Idle)
        end
        bedrock:Destroy()
        if LocalPlayer.Character then
            LocalPlayer.Character.Animate.Enabled = true
            local root = LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
            if root then
                root.Anchored = false
            end
        end
        text:Destroy()
        -- æ¸çåé
        hum = nil
        root = nil
        size = nil
        events = nil
        eat = nil
        grab = nil
        sell = nil
        sendTrack = nil
        chunk = nil
        radius = nil
        -- éç½®çº¿ç¨åé
        autoFarmThread = nil
    end)()
end

local function getMapName()
    -- æ¹æ³1: æ£æ¥ReplicatedStorageä¸­çServerSettings
    local ServerSettings = ReplicatedStorage:FindFirstChild("ServerSettings")
    if ServerSettings then
        if ServerSettings:FindFirstChild("MapName") then
            return ServerSettings.MapName.Value, "ServerSettings.MapName"
        elseif ServerSettings:FindFirstChild("CurrentMap") then
            return ServerSettings.CurrentMap.Value, "ServerSettings.CurrentMap"
        end
    end
    -- æ¹æ³2: æ£æ¥å·¥ä½åºä¸­çMapæä»¶å¤¹
    local mapFolder = Workspace:FindFirstChild("Map")
    if mapFolder then
        return mapFolder.Name, "Workspace.Map.Name"
    end
    -- æ¹æ³3: æ£æ¥å·¥ä½åºä¸­çä¸»è¦é¨å
    local mainPart = Workspace:FindFirstChild("Baseplate") or Workspace:FindFirstChild("Terrain")
    if mainPart then
        return mainPart.Name, "Workspace.MainPart.Name"
    end
    -- æ¹æ³4: æ£æ¥æ¸¸æçPlaceId
    local placeId = game.PlaceId
    if placeId > 0 then
        return "Place ID: " .. placeId, "game.PlaceId"
    end
    -- æ¹æ³5: æ£æ¥å·¥ä½åºä¸­çæä»¶å¤¹ç»æ
    local folders = {}
    for _, child in ipairs(Workspace:GetChildren()) do
        if child:IsA("Folder") or child:IsA("Model") then
            table.insert(folders, child.Name)
        end
    end
    if #folders > 0 then
        return table.concat(folders, ", "), "Workspace.Folders"
    end
    -- æ¹æ³6: æ£æ¥LocalPlayerçå±æ§
    if LocalPlayer:FindFirstChild("CurrentMap") then
        return LocalPlayer.CurrentMap.Value, "LocalPlayer.CurrentMap"
    end
    -- æ æ³è·åå°å¾åç§°
    return "æªç¥å°å¾", "æ æ³è·å"
end

end
